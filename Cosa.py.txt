def area_poligono(vertices):
    """
    Calcula el área de un polígono dado por una lista de vértices.
    Cada vértice es una tupla (x, y).
    """
    n = len(vertices)
    area = 0

    for i in range(n):
        x1, y1 = vertices[i]
        x2, y2 = vertices[(i + 1) % n]
        area += x1 * y2 - x2 * y1

    return abs(area) / 2
